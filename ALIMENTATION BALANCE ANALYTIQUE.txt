COMMERCIAL04 : remise


SELECT 
    a.num_0 AS NUM_PIECE,

    DECODE(
        SUBSTR(a.bpcinv_0, 1, 1),
        'P', '71241000',  --ou 71290000
        'D', '71110000',  --ou 71190000
        '71110000'
    ) AS COMPTE,


    DECODE(SUBSTR(a.sivtyp_0, 1, 1), 'F', -1, 'A', 1, -1) AS SENS,

    c.cce_4 AS AXE_CENTRE,
    c.cce_7 AS AXE_ENTITE,
    c.cce_5 AS AXE_BLINE,
    c.cce_6 AS AXE_SITE,

    ABS(b.dtanot_0 * NVL(e.ratmlt_0, 1)) AS MONTANT,

    a.bpcinv_0 AS TIERS_CODE,
    NULL       AS ARTICLE_CODE,
    a.invdat_0 AS DATE_COMPTABLE,

    'COMMERCIAL04' AS SOURCE,
    'DETAIL REMISE'    AS TYPE_LIGNE

FROM sinvoicev a
JOIN svcrfoot b   ON a.num_0 = b.vcrnum_0
JOIN sinvoice e   ON e.num_0 = a.num_0
LEFT JOIN cptanalin c ON c.vcrnum_0 = a.num_0 
                     AND c.vcrlin_0 = 0

WHERE a.invdat_0 >= TO_DATE('01/01/2026', 'DD/MM/YYYY')
  AND a.invdtaamt_1 <> 0
  AND b.numdta_0 = 1
  AND SUBSTR(a.sivtyp_0, 1, 1) IN ('F', 'A');

COMMERCIAL06 AVOIR FACTURE


SELECT 
    x.num_0 as NUM_PIECE,
    x.accdat_0 as DATE_COMPTABLE,
    x.bpr_0 as TIERS_CODE,
    y.itmref_0 ARTICLE_CODE,


    ABS(y.qty_0 * y.netpri_0 * ratmlt_0) AS MONTANT ,

    DECODE(SUBSTR(x.gte_0, 1, 1), 'F', 1, 'A', -1) AS SENS, 


    c.cce_4 as AXE_CENTRE, c.cce_5 as AXE_BLINE,
    c.cce_6 as AXE_SITE, c.cce_7 as AXE_ENTITE, 

    DECODE(SUBSTR(x.bpr_0, 1, 1), 
           'P', '71241000',  -- ou 71290000 si compte RRR
           'D', '71110000',  -- ou 71190000 si compte RRR
           '71110000'
    ) AS COMPTE,

    'COMMERCIAL06' AS SOURCE,
    'DETAIL AVOIR FACT'    AS TYPE_LIGNE
        
FROM sinvoice x
JOIN sinvoiced y ON x.num_0 = y.num_0
JOIN itmmaster t ON t.itmref_0 = y.itmref_0
LEFT JOIN cptanalin c ON c.vcrnum_0 = y.num_0
                     AND c.vcrlin_0 = y.sidlin_0

WHERE x.accdat_0 >= TO_DATE('01/01/2026','DD/MM/YYYY')      
AND substr(x.num_0,1,2)  LIKE 'AF%'
