WITH DailyAggregates AS (
    -- Étape 1 : Agrégation des mouvements par article et par date
    SELECT 
        s.ITMREF_0 AS Article, 
        CAST(s.IPTDAT_0 AS DATE) AS DateMouvement, 
        SUM(s.QTYSTU_0) AS QteJour
    FROM 
        STOJOU s
    INNER JOIN 
        ITMMASTER i ON i.ITMREF_0 = s.ITMREF_0
    WHERE 
        s.IPTDAT_0 >= DATE '2026-01-01'
        AND CAST(i.CREDAT_0 AS DATE) <= CAST(s.IPTDAT_0 AS DATE)
    GROUP BY 
        s.ITMREF_0, 
        CAST(s.IPTDAT_0 AS DATE)
),
Articles AS (
    -- Étape 2 : Liste distincte des articles concernés
    SELECT DISTINCT Article FROM DailyAggregates
),
Calendrier AS (
    -- Étape 3 : Génération d'un calendrier continu (du 01/01/2026 à aujourd'hui)
    SELECT DATE '2026-01-01' + LEVEL - 1 AS DateMouvement
    FROM DUAL
    CONNECT BY DATE '2026-01-01' + LEVEL - 1 <= TRUNC(SYSDATE)
),
ArticleCalendrier AS (
    -- Étape 4 : Produit cartésien article x date (toutes les dates pour chaque article)
    SELECT a.Article, c.DateMouvement
    FROM Articles a
    CROSS JOIN Calendrier c
),
Combined AS (
    -- Étape 5 : Jointure avec les mouvements réels, 0 si absent
    SELECT 
        ac.Article,
        ac.DateMouvement,
        NVL(d.QteJour, 0) AS QteJour
    FROM ArticleCalendrier ac
    LEFT JOIN DailyAggregates d
        ON d.Article = ac.Article
       AND d.DateMouvement = ac.DateMouvement
)
-- Étape 6 : Calcul du cumul (reprend automatiquement le cumul de la veille si QteJour = 0)
SELECT
    Article,
    DateMouvement,
    QteJour,
    SUM(QteJour) OVER (
        PARTITION BY Article
        ORDER BY DateMouvement
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS QuantiteCumulative
FROM Combined
ORDER BY 
    Article, 
    DateMouvement;
